
# Iniciar MongoDB: Esto levantará los servicios, incluyendo el contenedor de MongoDB.

    docker compose -f docker-compose.dev.yml up -d

# Accede al contenedor usando docker exec
    docker ps
# Accede al contenedor de MongoDB
    docker exec -it <container_name_or_id> /bin/bash
